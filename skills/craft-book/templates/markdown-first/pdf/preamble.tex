% --- Chapter openings ---
\makeatletter
\def\@makechapterhead#1{%
  \vspace*{20pt}%
  {\parindent\z@\raggedright\normalfont
   \ifnum\c@secnumdepth>\m@ne
     {\small\color{teal}\textsc{\@chapapp\ \thechapter}}\par\vskip 8pt
   \fi
   {\Huge\color{navy}#1\par}\vskip 9pt
   {\color{gold}\rule{0.23\linewidth}{1.2pt}}\par\vskip 23pt}}

% --- Title page: pandoc calls \maketitle; we replace it with a designed page ---
\renewcommand{\maketitle}{%
  \begin{titlepage}
  \thispagestyle{empty}
  \centering
  \vspace*{1.0in}
  \ifdefined\booktagline{\small\color{teal}\textsc{\booktagline}\par}\fi
  \vspace{0.45in}
  {\fontsize{32}{38}\selectfont\color{navy}\textsc{\@title}\par}
  \vspace{0.28in}
  \ifdefined\booksubtitle{\Large\itshape\booksubtitle\par}\fi
  \vspace{0.5in}
  {\color{gold}\rule{0.55\linewidth}{1.2pt}\par}
  \vspace{0.45in}
  {\large\@author\par}
  \vfill
  {\small\@date\par}
  \end{titlepage}}
\makeatother
